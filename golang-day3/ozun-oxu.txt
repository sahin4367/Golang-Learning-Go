package main 

import ( 
    "fmt"
)

func bolme( a , b float64 ) (float64 , string) {
    if b == 0 {
        return 0 // "0- a bolmek  olmaz"~!
    }
    return a / b // "Success-dir~!"
}

func main() {
    fmt.Println(bolme(42,6))
}

onda biz Variadic funk istifade edirik! (...) bu 3 noqte ile isare olunur!

func SumNum(number ...int) int  {
    sum := 0
    for _, n := range number {
        sum += n
    }

    return sum
}

Bu o demekdir ki meselen case 1: , case 2: , case 3: , ve s/
SWITCH -- MULTIcase VAR .
Bu o  demekdir ki demeli case 1,2,3: , case 4,5 ve s/

package main 
import (
    'fmt'
)

func main(){
    bal := 88

    switch bal {
        case 56,67:
            fmt.Println("Bu ballar 88 deyil!")
        case 75,85,88:
            fmt.Println("Bu ballar asagi yuxari 88-dir!")
        defoult :
            fmt.Println("Eks halda sen sagol kesildin.")
    }
}

package main 
import ("fmt")

func main(){
    a := 3 
    switch a {
        case 1:
            fmt.Println("a 1e beraberdir.")
        case 3: 
            fmt.Println("a 3e beraberdir.")
        defoult:
            fmt.Println("a ne 1 nede 3 dur!")
    }
}

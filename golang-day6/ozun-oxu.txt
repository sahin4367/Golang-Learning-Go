    if f  < 0 {
        return float64(-f)

    return float64(f)

Pointer receivers - Aid numunelre izahllar:
type Nums struct {
	a, b float64
    //Burda bir type yaradiriq ele bil a ve b reqemlerini structile
}

func (c Nums) Abs() float64 {
	return math.Sqrt(c.a*c.a + c.b*c.b)
    //Daha sonra bir func yaradiriq Abs() ve onun c deyisenine Hasni type de oldugunu deyirik
    //ve bize float64 qaytaracaq(tipi bu olacaq), esas returun da riyazi  _/a^2+b^2 qaytaracaq! 
}

func (c *Nums) PointFunc(d float64) {
	c.a = c.a * d
	c.b = c.b * d
    //Bu ne demekdir ? 
    //c *Nums {bu Pointer demkdir - menasi odur ki orjinal struct deyisecek } , bele olmayanda ise
    //sadece Copy olaraq gedir type struct deyisilimir~!
    //burda PointFunc funk yaradir demeli d deyiseni teyin olunur okey~!
    //asagda maine baxsaq c := Nums{6,8} olsa bele o struct deyisecek 
    olacaq - {60,80} buna gorede Pointer bu isi gorud daha sonra 
    //esas func ise dusecek Abs() {bize returunu vercek }
    tebbi ki biz onu main funck cagiraraq print elesesk ~!
}

func main() {
	c := Nums{6, 8}
	c.PointFunc(10)
	fmt.Println(c.Abs())
}

INTERFASE : 
Nil sohbeti  demeli , qayda var ki interfase(value ve type) saxlayir bilirik.
Interface yalnız o zaman nil-dir ki,
type və value hər ikisi nil olsun.type == nil ve value==nil okey onda INTERFASE == nil olacaq~!
||--En
An interface value in Go is nil only when both its type and value are nil. 
If the interface holds a nil concrete value but has a type, the interface itself is not nil.
||--Aze
Go-da interface yalnız o zaman nil olur ki, həm type, həm də value nil olsun. 
Əgər interface daxilində type varsa, amma value nil-dirsə, interface özü nil hesab olunmur.
[
    BEZEN ELE OLUR KI DEMELI , MESELEN (nil,nil) olur interfacesde basa dusdun~!
    bu TAM INTERFASE adlanir. Yeni interface tam bosdur.
]

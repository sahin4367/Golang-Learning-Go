Golangda / Variable / Constants /
\--Variable demeli (var)
tutagki var x int | yaziram daha snra calisidriam mene 0 qaytaracaq , false ve ya null olmaz !
var - ile teyin olunan deyerler istenilen vaxt kodda deyisdireile biler
var da meselen x := 3 yazmaq mumkundur! 
\--Constants(const)
const ile teyin olunan deyerler daimidir,  deyismir! 
Golangda const-u teyin ederek ona mutleq deyer vermelisen , bos buraxmaq olmaz!
const da ise x := 3 X - mumkun deyil!!
\-- VARIABLE  NEDIR ? 
Variable(deyisen) demekdir ! 
variable komputere verielen melumarlari yadda saxlamq ucun onlara verilen addir!
meselen biz 5+10 yaziriq , komp bunu nece hesablair arxada eslinde 5 i bir qutuya qoyur,
daha sonra 10 bir qutuya qoyur, ve bunlarin cemini de bir qutuya qoyur! BUNU DA ADLANDIRIR
Deyisenler olmasaydi komputer  hec neyi yadda saxlamazdi!
\-deyisinlerin (ADI,DEYERI,TIPI) olur!
[VARIABLE] - var num  = 5 {num burda addir, 5 deyerdir , int ise bunun tipidir}.
--------------------------------------------------------------------------------------------
package main
import "fmt"

func main(){
    var a int = 20
    a = 12
    fmt.Print(a)
    //burda artiq demeli a = 20 olacaq !
}
////////////////////////////////////////////////////////////////////////////////////////////
package main
import "fmt"

func main(){
    const a int = 20
    a = 12 //bura qizaracaq
    fmt.Print(a)
    //burda artiq kod islemeyecek xeta verecek 
}

Golangda - Data type 
\-- golangda data tipler bir sira qruplara bolunur demeli,
komputer ucun her sey esllinde 0 ve 1 den ibaretdir!
proqramci ucun ise o 0 ve 1 -in ne mena verrdiyi onemlidir
\\--Golangda\int\float'32'64\\string\bool\
package main 
import "fmt"
func main(){
    var a = 5 // var a int = 5 |Golang birincinde int kimi gorur!
    var b string = "Shahin"
    var c bool = true
    var d float32 = 5.32

    fmt.Printf("a-nin tipi :  %T\n" , a) // bununla a deyiseninin tipini tapiriQ~!
    fmt.Println(a,b,c,d)
}
\-Eger tam rahat tez tapmaq istiyirsen ki verilen deyisenin tipi nedir-/Onda\
reflect - paketini endiririk yeni 
package main

import (
    "fmt"
    "reflect"
)

func main(){
    var c = false
    t := reflect.TypeOf(c)
    fmt.Print(t) // bu bize bool tipini qaytaraca
}
\-Type Conversions-/Tip cevrilmeleri\
package main 
import (
    "fmt"
    "reflect"
)

func main(){
    var a int = 5
    var f float = float(a)
    var i uint = uint(f)

    s := reflect.TypeOf(i)
    fmt.Print("i-nin tipi  : " , s) || sen tip cevrilmelri bax bu sekilde ede bilersen! }
